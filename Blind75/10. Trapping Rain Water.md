/**
 * LC 42 - Trapping Rain Water | Hard | Two Pointers
 *
 * RECOGNIZE: water above a bar depends on max wall on each side
 * KEY IDEA:  water[i] = min(maxLeft, maxRight) - height[i]
 * APPROACH:  two pointers; process the side with the smaller max,
 *            because that side is the bottleneck
 * COMPLEXITY: O(n) time, O(1) space (prefix/suffix arrays: O(n) space)
 * TRAPS:     update max BEFORE computing water; loop on pointers,
 *            not on maxes; guard null / length < 3
 * WHY SAFE:  if leftMax <= rightMax, true right max is at least
 *            rightMax, so leftMax is the limit
 * REVISIT:   2026-10-23
 */


```java
public int trap(int[] height) {
    if(height == null || height.length < 3)
        return 0;

    int left = 0;
    int right = height.length-1;
    int leftMax = height[left];
    int rightMax = height[right];
    int totalArea = 0;

    while(left < right) {
        if(leftMax < rightMax) {
            left++;
            leftMax = Math.max(leftMax, height[left]);
            totalArea += leftMax - height[left];
        } else {
            right--;
            rightMax = Math.max(rightMax, height[right]);
            totalArea += rightMax - height[right];
        }
    }
    return totalArea;
}


public int trap(int[] height) {
    if(height == null || height.length < 3)
        return 0;

    int[] leftMax = new int[height.length];
    int[] rightMax = new int[height.length];
    int currMax = 0;
    leftMax[0] = 0;
    for(int i=1; i<height.length; i++) {
        currMax = Math.max(currMax, height[i-1]);
        leftMax[i] = currMax;
    }

    int currRmax = 0;
    rightMax[height.length-1] = 0;
    for(int i=height.length-2; i>=0; i--) {
        currRmax = Math.max(currRmax, height[i+1]);
        rightMax[i] = currRmax;
    }

    int totalArea = 0;
    for(int k=0; k<height.length; k++) {
        totalArea += Math.max((Math.min(leftMax[k], rightMax[k]) - height[k]), 0);
    }

    return totalArea;
}
```
